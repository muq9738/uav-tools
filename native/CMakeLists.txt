cmake_minimum_required(VERSION 3.20)
# Share the desktop release version and reconfigure when the manifest changes.
set(OSGB_APP_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/../app/mygo.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${OSGB_APP_MANIFEST}")
file(READ "${OSGB_APP_MANIFEST}" OSGB_APP_MANIFEST_JSON)
string(JSON OSGB_RELEASE_VERSION GET "${OSGB_APP_MANIFEST_JSON}" version)
project(osgb_tiles_native VERSION "${OSGB_RELEASE_VERSION}" LANGUAGES C CXX)
configure_file(cmake/product_metadata.hpp.in generated/product_metadata.hpp @ONLY)
# 构建树从实际链接目录解析 @rpath，不写死 Intel/ARM Homebrew 前缀。
set(CMAKE_BUILD_RPATH_USE_LINK_PATH TRUE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(OSGB_BUILD_DIAGNOSTIC_TOOLS "Build optional source LOD audit/render tools" OFF)
set(OSGB_OSG_COMPONENTS osg osgDB)
if(OSGB_BUILD_DIAGNOSTIC_TOOLS)
  list(APPEND OSGB_OSG_COMPONENTS osgViewer)
endif()
find_package(OpenSceneGraph 3.6 REQUIRED COMPONENTS ${OSGB_OSG_COMPONENTS})
find_package(PNG REQUIRED)
find_package(LibXml2 REQUIRED)
find_package(Threads REQUIRED)
option(BUILD_TESTING "构建原生回归测试" ON)
include(cmake/Dependencies.cmake)
include(cmake/Codecs.cmake)
add_library(meshoptimizer_codec STATIC third_party/meshoptimizer/allocator.cpp third_party/meshoptimizer/indexcodec.cpp third_party/meshoptimizer/vertexcodec.cpp)
target_include_directories(meshoptimizer_codec PUBLIC third_party/meshoptimizer)
# 实验候选可独立链接；默认转换入口不调用它。
add_library(osgb_compressed_image_candidate STATIC src/compressed_image.cpp)
target_include_directories(osgb_compressed_image_candidate PUBLIC include)
add_library(osgb_tiles_core src/image.cpp src/glb.cpp src/orthophoto.cpp src/convert.cpp src/output_validation.cpp src/spatial.cpp src/meshopt.cpp src/runtime.cpp src/checkpoint.cpp src/top.cpp src/preview.cpp src/draco.cpp src/basis.cpp)
target_include_directories(osgb_tiles_core PUBLIC include third_party ${OPENSCENEGRAPH_INCLUDE_DIRS})
target_link_libraries(osgb_tiles_core PUBLIC ${OPENSCENEGRAPH_LIBRARIES} PNG::PNG LibXml2::LibXml2 Threads::Threads PROJ::proj meshoptimizer_codec osgb_compressed_image_candidate draco::draco basis_native)
target_include_directories(osgb_tiles_core PRIVATE ${draco_SOURCE_DIR}/src ${CMAKE_BINARY_DIR}
  "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(osgb_tiles_core PUBLIC GL_SILENCE_DEPRECATION)
add_dependencies(osgb_tiles_core generate_proj_db)
if(MSVC)
  target_compile_options(osgb_tiles_core PRIVATE /W4 /utf-8 /bigobj)
  target_compile_options(osgb_compressed_image_candidate PRIVATE /W4 /utf-8)
  target_compile_definitions(osgb_tiles_core PUBLIC NOMINMAX)
  target_link_libraries(osgb_tiles_core PUBLIC psapi)
  add_compile_options(/utf-8 /bigobj)
else()
  target_compile_options(osgb_tiles_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(osgb_compressed_image_candidate PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(osgb-tiles src/main.cpp)
target_link_libraries(osgb-tiles PRIVATE osgb_tiles_core)
if(MSVC)
  target_sources(osgb-tiles PRIVATE cmake/utf8.manifest)
endif()
if(OSGB_BUILD_DIAGNOSTIC_TOOLS)
add_executable(osgb-source-lod-audit tools/audit_source_lod.cpp)
target_link_libraries(osgb-source-lod-audit PRIVATE osgb_tiles_core)
add_executable(osgb-source-lod-render tools/render_source_lod.cpp)
target_link_libraries(osgb-source-lod-render PRIVATE osgb_tiles_core)
if(APPLE)
  target_link_libraries(osgb-source-lod-render PRIVATE "-framework OpenGL")
else()
  find_package(OpenGL REQUIRED)
  target_link_libraries(osgb-source-lod-render PRIVATE OpenGL::GL)
endif()
find_path(CGAL_INCLUDE_DIR CGAL/Polygon_mesh_processing/distance.h)
if(CGAL_INCLUDE_DIR)
  add_executable(osgb-lod-error-probe tools/measure_lod_error.cpp)
  target_include_directories(osgb-lod-error-probe PRIVATE ${CGAL_INCLUDE_DIR})
  target_compile_definitions(osgb-lod-error-probe PRIVATE CGAL_NO_GMP=1 CGAL_NO_MPFR=1)
  target_link_libraries(osgb-lod-error-probe PRIVATE osgb_tiles_core)
endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(osgb-tiles-tests tests/test_main.cpp)
  target_link_libraries(osgb-tiles-tests PRIVATE osgb_tiles_core)
  target_compile_definitions(osgb-tiles-tests PRIVATE TEST_PROJ_DATA="${proj_BINARY_DIR}/data")
  add_test(NAME native-conversion COMMAND osgb-tiles-tests)
  add_executable(osgb-texture-tests tests/texture_cases.cpp)
  target_link_libraries(osgb-texture-tests PRIVATE osgb_tiles_core)
  add_test(NAME native-texture-compatibility COMMAND osgb-texture-tests)
  add_executable(osgb-output-validation-tests tests/output_validation_cases.cpp)
  target_link_libraries(osgb-output-validation-tests PRIVATE osgb_tiles_core)
  target_include_directories(osgb-output-validation-tests PRIVATE src)
  add_test(NAME native-output-validation COMMAND osgb-output-validation-tests)
  add_executable(osgb-spatial-tests tests/spatial_cases.cpp)
  target_link_libraries(osgb-spatial-tests PRIVATE osgb_tiles_core)
  add_test(NAME native-spatial-contract COMMAND osgb-spatial-tests)
  add_executable(osgb-compressed-image-tests tests/compressed_image_cases.cpp)
  target_link_libraries(osgb-compressed-image-tests PRIVATE osgb_tiles_core)
  target_compile_definitions(osgb-compressed-image-tests PRIVATE TEST_BC_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/bc-candidate-vectors.json")
  add_test(NAME native-compressed-image-candidate COMMAND osgb-compressed-image-tests)
  add_executable(osgb-checkpoint-tests tests/checkpoint_cases.cpp)
  target_include_directories(osgb-checkpoint-tests PRIVATE src)
  target_link_libraries(osgb-checkpoint-tests PRIVATE osgb_tiles_core)
  add_test(NAME native-checkpoint COMMAND osgb-checkpoint-tests)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME native-checkpoint-cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/pause_resume_cli.py $<TARGET_FILE:osgb-tiles> $<TARGET_FILE:osgb-checkpoint-tests>)
  endif()
  add_executable(osgb-codec-tests tests/codec_cases.cpp)
  target_link_libraries(osgb-codec-tests PRIVATE osgb_tiles_core)
  target_include_directories(osgb-codec-tests PRIVATE src ${draco_SOURCE_DIR}/src ${CMAKE_BINARY_DIR})
  add_test(NAME native-codecs COMMAND osgb-codec-tests)
  add_executable(osgb-top-tests tests/top_cases.cpp)
  target_link_libraries(osgb-top-tests PRIVATE osgb_tiles_core)
  target_include_directories(osgb-top-tests PRIVATE src)
  add_test(NAME native-top-rebuild COMMAND osgb-top-tests)
  add_executable(osgb-preview-tests tests/preview_cases.cpp)
  target_link_libraries(osgb-preview-tests PRIVATE osgb_tiles_core)
  target_include_directories(osgb-preview-tests PRIVATE src)
  add_test(NAME native-mesh-preview COMMAND osgb-preview-tests)
  add_executable(osgb-reference-tests tests/reference_cases.cpp)
  target_link_libraries(osgb-reference-tests PRIVATE osgb_tiles_core)
  add_test(NAME native-reference-integrity COMMAND osgb-reference-tests)
  add_executable(osgb-geometry-tests tests/geometry_compatibility.cpp)
  target_link_libraries(osgb-geometry-tests PRIVATE osgb_tiles_core)
  add_test(NAME native-geometry-compatibility COMMAND osgb-geometry-tests)
  add_executable(osgb-pagedlod-split-tests tests/pagedlod_split_cases.cpp)
  target_link_libraries(osgb-pagedlod-split-tests PRIVATE osgb_tiles_core)
  add_test(NAME native-pagedlod-split COMMAND osgb-pagedlod-split-tests)
  add_executable(osgb-quantization-tests tests/quantization_cases.cpp)
  target_link_libraries(osgb-quantization-tests PRIVATE osgb_tiles_core)
  add_test(NAME native-quantization COMMAND osgb-quantization-tests)
  find_program(OSGB_NODE_EXECUTABLE node)
  if(OSGB_NODE_EXECUTABLE)
    add_test(NAME native-glb-accessor-oracle COMMAND ${OSGB_NODE_EXECUTABLE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/glb-accessors.test.mjs ${CMAKE_CURRENT_SOURCE_DIR}/tests/oriented-triangles.test.mjs)
  endif()
  # 独立 oracle 的图片兼容回归：PNG 由测试自己的 zlib 生成，JPEG 由 turbojpeg 生成与解码，
  # 生产代码仍只用 libpng，不引入新依赖。turbojpeg 只是测试依赖：找不到时跳过该测试并明确告警，
  # 不让其他平台因为缺少测试依赖而无法构建。
  find_package(ZLIB REQUIRED)
  find_path(TURBOJPEG_INCLUDE_DIR turbojpeg.h)
  find_library(TURBOJPEG_LIBRARY turbojpeg)
  if(TURBOJPEG_INCLUDE_DIR AND TURBOJPEG_LIBRARY)
    add_executable(osgb-texture-oracle-tests tests/texture_oracle_cases.cpp)
    target_include_directories(osgb-texture-oracle-tests PRIVATE ${TURBOJPEG_INCLUDE_DIR})
    target_link_libraries(osgb-texture-oracle-tests PRIVATE osgb_tiles_core ZLIB::ZLIB ${TURBOJPEG_LIBRARY})
    target_compile_definitions(osgb-texture-oracle-tests PRIVATE
      TEST_OSGB_TEMPLATE="${CMAKE_CURRENT_SOURCE_DIR}/../data/test/test.osgb")
    add_test(NAME native-texture-oracle COMMAND osgb-texture-oracle-tests --strict)
  else()
    message(WARNING "未找到 turbojpeg（libjpeg-turbo）：跳过图片兼容独立 oracle 回归 "
      "tests/texture_oracle_cases.cpp。安装提供 turbojpeg.h 与 libturbojpeg 的 libjpeg-turbo 后重新配置即可启用。")
  endif()
endif()
# 仅修正本工程生成文件里的 Homebrew 占位加载路径，不修改系统动态库。
if(APPLE)
  set(OSGB_BUILD_RPATHS)
  foreach(library IN LISTS OPENSCENEGRAPH_LIBRARIES PNG_LIBRARIES LIBXML2_LIBRARIES SQLite3_LIBRARY SQLite3_LIBRARIES TURBOJPEG_LIBRARY)
    if(IS_ABSOLUTE "${library}" AND "${library}" MATCHES "\\.dylib$")
      get_filename_component(directory "${library}" DIRECTORY)
      list(APPEND OSGB_BUILD_RPATHS "${directory}")
    endif()
  endforeach()
  list(REMOVE_DUPLICATES OSGB_BUILD_RPATHS)
  foreach(target osgb-tiles osgb-source-lod-audit osgb-source-lod-render osgb-lod-error-probe osgb-tiles-tests osgb-texture-tests osgb-output-validation-tests osgb-spatial-tests osgb-compressed-image-tests osgb-checkpoint-tests osgb-reference-tests osgb-geometry-tests osgb-texture-oracle-tests osgb-pagedlod-split-tests osgb-quantization-tests osgb-top-tests osgb-preview-tests osgb-codec-tests)
    if(TARGET ${target})
      set_property(TARGET ${target} PROPERTY BUILD_RPATH_USE_LINK_PATH TRUE)
      set_property(TARGET ${target} APPEND PROPERTY BUILD_RPATH "${OSGB_BUILD_RPATHS}")
      add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND}
        -DBINARY=$<TARGET_FILE:${target}> -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/macos-rpath.cmake)
    endif()
  endforeach()
endif()
